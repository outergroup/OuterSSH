cmake_minimum_required(VERSION 3.24)
project(OuterSSHHelperWindows LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "This helper requires Windows")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(OuterSSHHelper.cmake)
outerssh_add_windows_helper(outerssh-helper Main.c)
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
    add_test(NAME named-pipe-bridge COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_bridge.py" $<TARGET_FILE:outerssh-helper>)
    set_tests_properties(named-pipe-bridge PROPERTIES TIMEOUT 60)
endif()
install(TARGETS outerssh-helper RUNTIME DESTINATION .)
