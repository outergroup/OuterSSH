cmake_minimum_required(VERSION 3.24)
project(OuterSSH VERSION 0.1 LANGUAGES C)

if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$" AND NOT WIN32 AND NOT ANDROID)
    message(FATAL_ERROR "This build targets macOS, Linux, Windows, and Android; use the Xcode target on iOS")
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
# GCC does not recognize the navigation-only #pragma mark directives.
add_compile_options("$<$<C_COMPILER_ID:GNU>:-Wno-unknown-pragmas>")
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    # MSVC warns about #pragma mark and standard C flexible array members.
    add_compile_options(/experimental:c11atomics /utf-8 /wd4068 /wd4200)
endif()
option(OUTERSSH_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(OUTERSSH_BUILD_CLI "Build the outerssh command-line tool" ${PROJECT_IS_TOP_LEVEL})
option(OUTERSSH_BUILD_TESTS "Build the OuterSSH tests" ${PROJECT_IS_TOP_LEVEL})
option(OUTERSSH_BUILD_BENCHMARK "Build the streaming throughput benchmark (Linux)" OFF)
if(OUTERSSH_SANITIZE)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        message(FATAL_ERROR "OUTERSSH_SANITIZE requires GCC or Clang")
    endif()
    add_compile_options("$<$<COMPILE_LANGUAGE:C>:-fsanitize=address,undefined;-fno-omit-frame-pointer>")
    add_link_options("$<$<LINK_LANGUAGE:C>:-fsanitize=address,undefined>"
        "$<$<LINK_LANGUAGE:Swift>:-sanitize=address,undefined>")
endif()

find_package(Threads REQUIRED)
if(WIN32)
    set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()
option(OUTERSSH_WINDOWS_CRYPTO "Use Windows CNG instead of OpenSSL" ${WIN32})
option(OUTERSSH_ANDROID_CRYPTO "Use Android platform cryptography instead of OpenSSL" OFF)
option(OUTERSSH_APPLE_CRYPTO "Use CryptoKit and Security instead of OpenSSL" ${APPLE})
if(OUTERSSH_ANDROID_CRYPTO)
    if(NOT ANDROID OR OUTERSSH_WINDOWS_CRYPTO)
        message(FATAL_ERROR "Android cryptography requires Android and cannot be combined with Windows CNG")
    endif()
    add_compile_definitions(OUTERSSH_ANDROID_CRYPTO=1)
    set(OUTERSSH_CRYPTO_LIBRARIES "")
elseif(OUTERSSH_WINDOWS_CRYPTO)
    if(NOT WIN32)
        message(FATAL_ERROR "Windows CNG requires Windows")
    endif()
    add_compile_definitions(OUTERSSH_WINDOWS_CRYPTO=1)
    set(OUTERSSH_CRYPTO_LIBRARIES bcrypt crypt32)
elseif(OUTERSSH_APPLE_CRYPTO)
    if(NOT APPLE)
        message(FATAL_ERROR "Apple cryptography requires macOS")
    endif()
    if(NOT CMAKE_GENERATOR MATCHES "Ninja|Xcode")
        message(FATAL_ERROR "Apple cryptography is written in Swift, which CMake builds only with "
            "the Ninja or Xcode generator. Pass -G Ninja, or -DOUTERSSH_APPLE_CRYPTO=OFF to use OpenSSL.")
    endif()
    enable_language(Swift)
    add_compile_definitions("$<$<COMPILE_LANGUAGE:C>:OUTERSSH_APPLE_CRYPTO=1>")
    set(OUTERSSH_CRYPTO_LIBRARIES "-framework CryptoKit" "-framework Foundation" "-framework Security")
else()
    find_package(OpenSSL 3.0 REQUIRED)
    set(OUTERSSH_CRYPTO_LIBRARIES OpenSSL::Crypto)
endif()
if(WIN32)
    add_compile_definitions(_WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX
        _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
    set(OUTERSSH_RUNTIME_SOURCES platform/OTQueueWindows.c platform/OTSocketWindows.c)
elseif(APPLE)
    set(OUTERSSH_RUNTIME_SOURCES platform/OTQueueApple.c)
else()
    set(OUTERSSH_RUNTIME_SOURCES platform/OTQueueLinux.c)
endif()
add_library(OuterSSHRuntime STATIC ${OUTERSSH_RUNTIME_SOURCES})
target_link_libraries(OuterSSHRuntime PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(OuterSSHRuntime PUBLIC ws2_32 bcrypt)
elseif(APPLE)
    target_link_libraries(OuterSSHRuntime PUBLIC objc "-framework Security")
endif()

find_package(ZLIB REQUIRED)

file(GLOB OUTERSSH_SOURCES CONFIGURE_DEPENDS "*.c" "olibssh/*.c" "olibssh/libssh2/*.c" "llhttp/*.c")
list(FILTER OUTERSSH_SOURCES EXCLUDE REGEX "/blowfish\\.c$")
set(LLHTTP_SOURCES llhttp/api.c llhttp/http.c llhttp/llhttp.c)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    set_source_files_properties(${LLHTTP_SOURCES} PROPERTIES COMPILE_OPTIONS "/wd4100;/wd4152;/wd4244")
else()
    set_source_files_properties(${LLHTTP_SOURCES} PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter")
endif()
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    # Retained libssh2 sources compile empty or with dead branches for backends this build omits.
    file(GLOB LIBSSH2_SOURCES CONFIGURE_DEPENDS "olibssh/libssh2/*.c")
    set_source_files_properties(${LIBSSH2_SOURCES} PROPERTIES COMPILE_OPTIONS "/wd4206;/wd4702")
endif()
add_library(OuterSSH STATIC ${OUTERSSH_SOURCES})
if(OUTERSSH_ANDROID_CRYPTO)
    target_compile_definitions(OuterSSH PUBLIC OUTERSSH_ANDROID_CRYPTO=1)
    target_include_directories(OuterSSH PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/platform/android")
    target_sources(OuterSSH PRIVATE platform/android/AndroidCryptoBridge.c)
endif()
add_library(OuterSSH::OuterSSH ALIAS OuterSSH)
if(OUTERSSH_WINDOWS_CRYPTO)
    target_compile_definitions(OuterSSH PUBLIC OUTERSSH_WINDOWS_CRYPTO=1)
endif()
if(OUTERSSH_APPLE_CRYPTO)
    target_sources(OuterSSH PRIVATE AppleCrypto.swift)
    set_target_properties(OuterSSH PROPERTIES Swift_MODULE_NAME OuterSSHAppleCrypto Swift_LANGUAGE_VERSION 6)
    target_compile_options(OuterSSH PRIVATE "$<$<COMPILE_LANGUAGE:Swift>:-DOUTERSSH_APPLE_CRYPTO>")
endif()
target_include_directories(OuterSSH
    PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include/OuterSSH"
        "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(OuterSSH PRIVATE /W4 /WX /wd4505)
else()
    target_compile_options(OuterSSH PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror;-Wno-unused-function>")
endif()
target_link_libraries(OuterSSH PUBLIC OuterSSHRuntime
    PRIVATE ${OUTERSSH_CRYPTO_LIBRARIES} ZLIB::ZLIB Threads::Threads)

if(OUTERSSH_BUILD_CLI)
    add_executable(outerssh ../OuterSSHCLI/main.c ../OuterSSHCLI/CLIPlatform.c)
    target_include_directories(outerssh PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
    if(WIN32)
        set_target_properties(outerssh PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin"
            ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin"
            PDB_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
    endif()
    target_link_libraries(outerssh PRIVATE OuterSSH ${OUTERSSH_CRYPTO_LIBRARIES} ZLIB::ZLIB Threads::Threads)
    set(OUTERSSH_CLI_VERSION "development" CACHE STRING "Version that outerssh --version prints")
    target_compile_definitions(outerssh PRIVATE "$<$<COMPILE_LANGUAGE:C>:OUTERSSH_CLI_VERSION=\"${OUTERSSH_CLI_VERSION}\">")
    if(MSVC)
        target_compile_options(outerssh PRIVATE /W4 /WX /wd4324)
    else()
        target_compile_options(outerssh PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror>")
    endif()
    install(TARGETS outerssh RUNTIME DESTINATION bin)
endif()

install(TARGETS OuterSSH OuterSSHRuntime ARCHIVE DESTINATION lib)
install(DIRECTORY include/OuterSSH DESTINATION include FILES_MATCHING PATTERN "*.h")

if(OUTERSSH_BUILD_BENCHMARK)
    if(WIN32 OR OUTERSSH_SANITIZE OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
        message(FATAL_ERROR "The throughput benchmark requires an unsanitized Linux Release build")
    endif()
    add_executable(benchmark-outerssh ../Scripts/benchmark_outerssh.c)
    target_include_directories(benchmark-outerssh PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_SOURCE_DIR}/include/OuterSSH"
        "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
    target_compile_options(benchmark-outerssh PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror>")
    target_link_libraries(benchmark-outerssh PRIVATE OuterSSH ${OUTERSSH_CRYPTO_LIBRARIES})
    add_executable(benchmark-packet-crypto ../Scripts/benchmark_ssh_packet_crypto.c)
    target_include_directories(benchmark-packet-crypto PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
    target_compile_options(benchmark-packet-crypto PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror>")
    target_link_libraries(benchmark-packet-crypto PRIVATE OuterSSH ${OUTERSSH_CRYPTO_LIBRARIES})
endif()

if(OUTERSSH_BUILD_TESTS)
    enable_testing()
    add_executable(outerssh-http-rewriter-test ../Scripts/test_http_proxy_rewriter.c)
    target_link_libraries(outerssh-http-rewriter-test PRIVATE OuterSSHRuntime)
    target_include_directories(outerssh-http-rewriter-test PRIVATE include/OuterSSH)
    if(MSVC)
        target_compile_options(outerssh-http-rewriter-test PRIVATE /UNDEBUG /W4 /WX)
    else()
        target_compile_options(outerssh-http-rewriter-test PRIVATE "$<$<COMPILE_LANGUAGE:C>:-UNDEBUG;-Wall;-Wextra;-Werror>")
    endif()
    add_test(NAME http-rewriter COMMAND outerssh-http-rewriter-test)
    add_executable(outerssh-http-proxy-fixture ../Scripts/test_http_proxy_fixture.c)
    target_include_directories(outerssh-http-proxy-fixture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" include/OuterSSH olibssh/libssh2)
    target_link_libraries(outerssh-http-proxy-fixture PRIVATE OuterSSH ${OUTERSSH_CRYPTO_LIBRARIES})
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME http-proxy COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/../Scripts/test_http_proxy.py" $<TARGET_FILE:outerssh-http-proxy-fixture>)
        set_tests_properties(http-proxy PROPERTIES TIMEOUT 120)
    endif()
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        add_compile_options(/UNDEBUG)
    else()
        add_compile_options("$<$<COMPILE_LANGUAGE:C>:-UNDEBUG>")
    endif()
    add_subdirectory(../Scripts tests)
    add_executable(outerssh-initialization-test ../Scripts/test_outerssh_initialization.c OuterSSH.c SSHEvents.c SSHEndpointNames.c SSHBridgeBinary.c SSHProxyError.c)
    target_include_directories(outerssh-initialization-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/include/OuterSSH"
        "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
    target_link_libraries(outerssh-initialization-test PRIVATE OuterSSHRuntime)
    add_test(NAME initialization COMMAND outerssh-initialization-test)
    add_test(NAME initialization-failure COMMAND outerssh-initialization-test fail)
    add_executable(outerssh-library-test ../Scripts/test_outerssh_library.c)
    target_include_directories(outerssh-library-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include/OuterSSH" "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
    target_link_libraries(outerssh-library-test PRIVATE OuterSSH)
    add_test(NAME public-library COMMAND outerssh-library-test)
    add_executable(outerssh-integration-test ../Scripts/test_outerssh_integration.c)
    target_include_directories(outerssh-integration-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(outerssh-integration-test PRIVATE OuterSSH)
    add_executable(outerssh-authentication-test ../Scripts/test_outerssh_authentication.c)
    target_include_directories(outerssh-authentication-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_SOURCE_DIR}/include/OuterSSH" "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
    target_link_libraries(outerssh-authentication-test PRIVATE OuterSSH)
    add_executable(outerssh-shared-proxy-test ../Scripts/test_outerssh_shared_proxy.c)
    target_include_directories(outerssh-shared-proxy-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include/OuterSSH" "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
    target_link_libraries(outerssh-shared-proxy-test PRIVATE OuterSSH)
    if(NOT WIN32)
        add_executable(outerssh-test-bridge ../OuterSSHHelper/Main.c ../OuterSSHHelper/Helper.c SSHEndpointNames.c)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME socket-allowlists COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/../Scripts/test_socket_allowlists.py"
            $<TARGET_FILE:outerssh-test-bridge>)
        set_tests_properties(socket-allowlists PROPERTIES TIMEOUT 20)
        add_test(NAME endpoint-icon-service COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/../Scripts/test_endpoint_icon_service.py"
            $<TARGET_FILE:outerssh-test-bridge>)
        set_tests_properties(endpoint-icon-service PROPERTIES TIMEOUT 20)
        add_test(NAME endpoint-names-local COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/../Scripts/test_endpoint_names.sh")
        set_tests_properties(endpoint-names-local PROPERTIES TIMEOUT 120)
        foreach(test IN ITEMS outerssh-rekey-test outerssh-reactor-rekey-test)
            add_executable(${test} ../Scripts/test_olibssh_rekey.c)
            target_include_directories(${test} PRIVATE
                "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include/OuterSSH"
                "${CMAKE_CURRENT_SOURCE_DIR}/olibssh/libssh2")
            target_link_libraries(${test} PRIVATE OuterSSH ${OUTERSSH_CRYPTO_LIBRARIES})
        endforeach()
        target_compile_definitions(outerssh-reactor-rekey-test PRIVATE SSH_REKEY_REACTOR_TEST)
    endif()
endif()

if(OUTERSSH_WINDOWS_CRYPTO AND OUTERSSH_BUILD_TESTS)
    enable_testing()
    add_executable(outerssh-windows-crypto-test ../Scripts/test_windows_crypto.c)
    target_include_directories(outerssh-windows-crypto-test PRIVATE olibssh/libssh2)
    target_link_libraries(outerssh-windows-crypto-test PRIVATE OuterSSH)
    target_compile_options(outerssh-windows-crypto-test PRIVATE $<IF:$<C_COMPILER_ID:MSVC>,/UNDEBUG,-UNDEBUG>)
    add_test(NAME windows-crypto COMMAND outerssh-windows-crypto-test)
endif()
